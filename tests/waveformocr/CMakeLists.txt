cmake_minimum_required(VERSION 3.16)
project(WaveformOcrRegression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network Concurrent Xml Test)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SOURCE "${REPO}/src")
# Reuse the actual application targets; no test copies of parser/state/storage logic.
add_subdirectory("${SOURCE}/data" data)
add_subdirectory("${SOURCE}/models" models)
include("${REPO}/cmake/ElectronicATECore.cmake")
set(ARCHITECTURE_SOURCE_ROOT "${SOURCE}")
include("${REPO}/cmake/CheckArchitecture.cmake")
enable_testing()
set(OFFLINE_SUITE ocr)
include(../cmake/offline-policy.cmake)
add_executable(parser_regression parser_regression.cpp)
target_include_directories(parser_regression PRIVATE "${SOURCE}")
target_link_libraries(parser_regression PRIVATE ElectronicATEWaveformOcr)
add_test(NAME waveform_ocr_parser COMMAND parser_regression)
add_executable(waveform_ocr_regression waveform_ocr_regression.cpp "${SOURCE}/views/waveformocr/waveformocrpage.cpp")
target_include_directories(waveform_ocr_regression PRIVATE "${SOURCE}")
target_link_libraries(waveform_ocr_regression PRIVATE ElectronicATEWaveformOcr Qt6::Widgets Qt6::Test)
offline_add_qttest(waveform_ocr_state_ui_export waveform_ocr_regression -o -,txt)
set_tests_properties(waveform_ocr_state_ui_export PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
offline_finalize_tests()
